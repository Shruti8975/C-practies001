
// for loop to print numbers from 1 to 10 

#include<stdio.h>
int main()
{
    for(int i = 1 ; i <= 10 ; i++){
        printf("%d\n",i);
    }
    return 0 ;
}


// for loop to print number from 10 to 1 
#include<stdio.h>
int main()
{
    for(int i = 10 ; i >= 1; i--){
        printf("%d\n",i);
    }
    return 0;
}
